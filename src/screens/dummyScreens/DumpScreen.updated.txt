const getStringValue = (

    value: unknown

): string | null => {

    if (

        typeof value !== "string"

    ) {

        return null;

    }



    const trimmedValue =

        value.trim();



    return trimmedValue.length > 0

        ? trimmedValue

        : null;

};



const getBooleanValue = (

    value: unknown

): boolean => {

    if (

        typeof value === "boolean"

    ) {

        return value;

    }



    if (

        typeof value === "string"

    ) {

        return (

            value.toLowerCase() ===

            "true"

        );

    }



    return false;

};
